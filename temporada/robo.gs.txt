// Robô de e-mail da Terê House's — roda de graça no Google, na conta que recebe os avisos do Airbnb.
// A cada hora lê os e-mails novos de "Reserva confirmada" e "Reserva cancelada" e manda o valor para o app.
// (As datas o próprio app já busca no calendário do Airbnb.) Ele nunca inventa: se não achar o valor com
// certeza, deixa em branco e a reserva aparece como "valor a lançar".
//
// Instalar: cole este código em script.google.com (Novo projeto), salve, escolha "instalar" e clique em Executar.

const SEGREDO = "COLE_AQUI";
const NUVEM_URL = "https://rexifhjmtphbpyirajgq.supabase.co";
const NUVEM_CHAVE = "sb_publishable_VHk3k9180XGEVrmWKWC3OA_f0BxEzJQ";
const BUSCA = "from:airbnb.com newer_than:45d subject:(confirmada OR confirmed OR cancelada OR canceled OR cancelled)";

function instalar() {
  ScriptApp.getProjectTriggers().forEach((t) => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger("rodar").timeBased().everyHours(1).create();
  rodar();
}

// Só mostra o que está lendo, sem mandar nada (Registro de execução).
function ensaiar() {
  const avisos = [];
  Logger.log(JSON.stringify(lerEmails([], avisos).map((e) => e.dados), null, 1));
  Logger.log("Avisos: " + JSON.stringify(avisos));
}

function rodar() {
  if (!SEGREDO || SEGREDO === "COLE_AQUI") throw new Error("Copie o código pelo app (Ajustes › Airbnb): ele já vem com o segredo.");
  const props = PropertiesService.getScriptProperties();
  const vistos = JSON.parse(props.getProperty("VISTOS") || "[]");
  const avisos = [];
  const lidos = lerEmails(vistos, avisos);
  const r = UrlFetchApp.fetch(NUVEM_URL + "/rest/v1/rpc/ab_robo", {
    method: "post", contentType: "application/json", headers: { apikey: NUVEM_CHAVE }, muteHttpExceptions: true,
    payload: JSON.stringify({ p_segredo: SEGREDO, p_dados: { emails: lidos.map((e) => e.dados), avisos: avisos } }),
  });
  if (r.getResponseCode() >= 300) throw new Error("O app recusou (" + r.getResponseCode() + "): " + r.getContentText());
  const relato = JSON.parse(r.getContentText());
  // E-mail de reserva que ainda não apareceu no calendário volta na próxima rodada.
  const pendentes = relato.pendentes || [];
  lidos.filter((e) => pendentes.indexOf(e.dados.codigo) < 0).forEach((e) => vistos.push(e.id));
  props.setProperty("VISTOS", JSON.stringify(vistos.slice(-300)));
  Logger.log(JSON.stringify(relato));
}

function lerEmails(vistos, avisos) {
  const saida = [];
  GmailApp.search(BUSCA, 0, 50).forEach((th) => th.getMessages().forEach((msg) => {
    const id = msg.getId();
    if (vistos.indexOf(id) >= 0 || !/airbnb\.com/i.test(msg.getFrom())) return;
    const assunto = msg.getSubject() || "";
    const corpo = msg.getPlainBody() || "";
    const cancelada = /cancelad|cancell?ed/i.test(assunto);
    const confirmada = /confirmad|confirmed/i.test(assunto);
    if (!cancelada && !confirmada) return;
    const cod = /\b(HM[A-Z0-9]{6,12})\b/.exec(assunto + "\n" + corpo);
    if (!cod) { avisos.push("E-mail sem código de reserva: \"" + assunto + "\""); return; }
    const dados = { codigo: cod[1], cancelada: cancelada };
    if (confirmada && !cancelada) {
      const h = /[-–]\s*(.+?)\s+(chega|arrives)/i.exec(assunto);
      if (h) dados.hospede = h[1].trim().slice(0, 80);
      const valor = valorDoAnfitriao(corpo);
      if (valor != null) dados.valor = valor;
      else avisos.push("Reserva " + cod[1] + ": não achei com certeza o valor que o anfitrião recebe — lance à mão.");
    }
    saida.push({ id: id, dados: dados });
  }));
  return saida;
}

// Só aceita o valor quando vem logo depois de um rótulo de ganho do anfitrião e há um único valor possível.
function valorDoAnfitriao(corpo) {
  const re = /(voc[eê] (ganha|recebe|receber[aá])|seus ganhos|ganhos do anfitri[aã]o|you earn|your earnings|host payout)[^\d]{0,60}?R?\$\s*([\d.]+,\d{2})/gi;
  const achados = {};
  let m;
  while ((m = re.exec(corpo))) achados[m[3]] = true;
  const valores = Object.keys(achados);
  if (valores.length !== 1) return null;
  return Math.round(parseFloat(valores[0].replace(/\./g, "").replace(",", ".")) * 100);
}
